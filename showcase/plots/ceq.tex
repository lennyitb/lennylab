% Conclusion margin figure: the two ways to combine capacitors, each drawn
% between open terminals with its C_eq formula underneath. Series on the
% left (stacked, like Circuit 4), parallel on the right (like Circuit 3).
\begin{circuitikz}[x=1cm, y=1cm, font=\small]
  \ctikzset{bipoles/length=0.9cm, capacitors/scale=0.8}
  \tikzset{head/.style={font=\sffamily\bfseries\fontsize{8.5}{10}\selectfont,
      anchor=base, inner sep=0pt}}
  % Series
  \begin{scope}[shift={(0,0)}]
    \node[head] at (0.9,2.75) {Series};
    \draw (0.9,2.4) node[ocirc] {}
          to[C, l=$C_1$, o-] (0.9,1.2)
          to[C, l=$C_2$, -o] (0.9,0);
    \node[anchor=north] at (0.9,-0.3)
      {$\dfrac{1}{\Ceq} = \dfrac{1}{C_1} + \dfrac{1}{C_2}$};
    \node[anchor=north] at (0.9,-1.35)
      {$\Ceq = \dfrac{C_1 \cdot C_2}{C_1 + C_2}$};
  \end{scope}
  % Parallel
  \begin{scope}[shift={(3.3,0)}]
    \node[head] at (0.75,2.75) {Parallel};
    \draw (0.75,2.4) node[ocirc] {} -- (0.75,1.9);
    \draw (0,1.9) -- (1.5,1.9);
    \draw (0,1.9) to[C, l_=$C_1$] (0,0.5);
    \draw (1.5,1.9) to[C, l=$C_2$] (1.5,0.5);
    \draw (0,0.5) -- (1.5,0.5);
    \draw (0.75,0.5) node[circ] {} -- (0.75,0) node[ocirc] {};
    \node[circ] at (0.75,1.9) {};
    \node[anchor=north] at (0.75,-0.3) {$\Ceq = C_1 + C_2$};
  \end{scope}
\end{circuitikz}
