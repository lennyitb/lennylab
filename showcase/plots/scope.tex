% Part 4 scope figure: every PicoScope capture, a 4 x 2 grid that fills
% one page. Rows are Circuits 1-4, columns charging and discharging. The
% time axis of each row is in multiples of that circuit's nominal R*C, so
% the 1-tau grid lines line up down the page.
%
% Data: data/scope-<c|d><n>.dat (and -zoom.dat for the insets) and
% data/scope-values.tex, all written by make-scope-data.py. Traces are
% shifted down by the scope's 0.408 V zero offset.
%
% Inset scale against its panel (\scW = 232 pt for 5.65 tau, inset 88 pt
% for 0.42 tau): 232/5.65 = 41.1 pt per tau, 88/0.42 = 209.5 pt per tau,
% so x5.1 in time. Vertically 124 pt for 17.3 V (7.2 pt/V) against 54 pt
% for 2 V (27 pt/V), x3.8.
%
% Stopwatch times are from the Part IV tab of Phelan Lab 06 Data.xlsx, over
% the nominal R*C: 49.66/47, 104.5/100, 154/147, 32.07/31.97.
% Adjusted tau: (R || R_VM)/R = 9.598/10.598 = 0.9056.
%
% \scH sets the panel height; it is tuned so the figure and its caption
% fill \textheight (see the \typeout below when changing it).
\input{data/scope-values}

% Legend samples, drawn with the panels' own styles. (Only in the legend
% line: a \tikz nested inside an axis would pick up its coordinates.)
\newcommand{\scsample}[1]{\tikz[baseline=-0.6ex]\draw[#1] (0,0) -- (13pt,0);}
\newcommand{\taudot}{\tikz[baseline=-0.6ex]\fill[accent] (0,0) circle[radius=1.7pt];}
\newcommand{\swmark}{\tikz[baseline=-0.6ex]\draw[chartText, line width=0.5pt]
  (90:2.6pt) -- (210:2.6pt) -- (330:2.6pt) -- cycle;}

% \scpanel{c|d}{circuit}{x}{y}{extra axis options}{stopwatch time / R*C, or empty}
\newcommand{\scpanel}[6]{%
  \edef\scK{#1#2}%
  \edef\tauN{\csname scope-\scK-tau\endcsname}%
  \edef\lvl{\csname scope-\scK-level\endcsname}%
  \pgfmathsetmacro{\scLo}{\lvl-1}\pgfmathsetmacro{\scHi}{\lvl+1}%
  \edef\scTaus{\csname scope-\scK-taus\endcsname}%
  \edef\scErr{\csname scope-\scK-err\endcsname}%
  \if c#1%
    \def\scRef{13.59}\def\inanchor{south east}\def\iny{0.4}%
    \def\tagy{9.0}\def\taganchor{south east}%
  \else
    \def\scRef{15.01}\def\inanchor{north east}\def\iny{14.4}%
    \def\tagy{6.2}\def\taganchor{north east}%
  \fi
  \begin{axis}[scope, #5, name=p\scK, at={(#3,#4)}, anchor=north west]
    \draw[level] (axis cs:-0.3,\scRef) -- (axis cs:5.35,\scRef);
    \addplot[trace] table[x=x, y=v] {data/scope-\scK.dat};
    \draw[zoom line] (axis cs:0.79,\scLo) rectangle (axis cs:1.21,\scHi);
    \coordinate (\scK-btr) at (axis cs:1.21,\scHi);
    \coordinate (\scK-bbr) at (axis cs:1.21,\scLo);
    \coordinate (\scK-in) at (axis cs:5.22,\iny);
    \node[verdict, anchor=\taganchor] at (axis cs:5.22,\tagy)
      {\(\tau\) = \qty{\scTaus}{\second}
       \textperiodcentered\ \qty{\scErr}{\percent} vs \(R \cdot C\)};
  \end{axis}
  \begin{axis}[scope inset, name=i\scK, at={(\scK-in)}, anchor=\inanchor,
      ymin=\scLo, ymax=\scHi]
    \draw[adj line] (axis cs:0.9056,\scLo) -- (axis cs:0.9056,\scHi);
    \draw[rc line] (axis cs:1,\scLo) -- (axis cs:1,\scHi);
    \draw[tau level] (axis cs:0.79,\lvl) -- (axis cs:1.21,\lvl);
    \addplot[trace, line width=0.8pt] table[x=x, y=v] {data/scope-\scK-zoom.dat};
    \ifx\relax#6\relax\else
      \addplot[only marks, mark=triangle, mark size=2.6pt,
        mark options={chartText, line width=0.5pt}] coordinates {(#6,8.59)};
    \fi
    \fill[accent] (axis cs:\tauN,\lvl) circle[radius=1.7pt];
  \end{axis}
  \draw[zoom line] (i\scK.north west) -- (\scK-btr);
  \draw[zoom line] (i\scK.south west) -- (\scK-bbr);
}
